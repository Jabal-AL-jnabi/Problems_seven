vector <string> PrintEachWorldInString(string s1,string space)
{

    short pos=0;
    short count=0;
    string words;
    vector <string> VWords;
    while((pos=s1.find(space))!=s1.npos)
    {
        words=s1.substr(0,pos);
        VWords.push_back(words);
        s1.erase(0,pos+space.length());
    }
    
    if(s1!="")
    {
        VWords.push_back(s1);
    }
    return VWords;

}
void PrintVector(vector <string> VS1)

   for(string &s1:VS1)

       cout<<s1<<endl;
   }
}

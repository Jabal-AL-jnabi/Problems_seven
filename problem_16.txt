void PrintMatrix(int arr[3][3], short Rows, short Cols)
{
    for (short i = 0; i < Rows; i++)
    {
        for (short j = 0; j < Cols; j++)
        {
            cout << arr[i][j] << "    ";
        }
        cout << "\n";
    }
}
int CountNumINMatrix(int arr[3][3], short Rows, short Cols,int Num)
{
  int count=0;
    for (short i = 0; i < Rows; i++)
    {
        for (short j = 0; j < Cols; j++)
        {
            if(arr[i][j]==Num)
            {
                count++;
            }
        }
    }
 return count;
}
int CheckSparce (int arr[3][3], short Rows, short Cols)
{
 short MatrixSiz=Cols*Rows  ;

}

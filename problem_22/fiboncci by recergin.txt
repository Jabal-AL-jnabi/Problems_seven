void Fibonacci (int num,int x,int y)
{
    int rsult =0;
    
    if(num>0)
    {

    cout<<rsult<<"  ";
    x=y;
    y=rsult;
    rsult=0;
    Fibonacci(num-1,x,y);
    
    }
}

bool CheckVowel (char c1)
{
    c1=tolower(c1);
    return ((c1=='a')||(c1=='e')||(c1=='i')||(c1=='o')||(c1=='u'));
}
short CountVowel (string s1)
{
   short j=0;
   for(int i=0;i<s1.length();++i)
   {
       if(CheckVowel(s1[i]))
       {
          ++j;  
       }
   }
   return j;
}

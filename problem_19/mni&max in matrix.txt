#include <iostream>
#include <string>
#include <iomanip>

using namespace std;

int getrand(int from,int to)
{
        return rand() %(to-from+1) +from;
}
void FullMatrixWithRandomNumber (int Arra[3][3],int row,int col)
{
  for(int i=0;i<row;++i)  
  {
      for(int j=0;j<col;++j)
      {
          Arra[i][j]=getrand(1,100);
      }
  }  

}
void PrintMatrix(int arr[3][3], short Rows, short Cols)
{
    for (short i = 0; i < Rows; i++)
    {
        for (short j = 0; j < Cols; j++)
        {
            cout << setw(3) << arr[i][j] << "    ";
        }
        cout << "\n";
    }
}
int MaxNumberInMatrix(int arr[3][3], short Rows, short Cols)
{
    int max = arr[0][0];

    for (short i = 0; i < Rows; i++)
    {
        for (short j = 0; j < Cols; j++)
        {
            if (arr[i][j] > max)
            {
                max = arr[i][j];
            }
        }
    }

    return max;
}
int MinNumberInMatrix(int arr[3][3], short Rows, short Cols)
{
    int min = arr[0][0];

    for (short i = 0; i < Rows; i++)
    {
        for (short j = 0; j < Cols; j++)
        {
            if (arr[i][j] < min)
            {
                min = arr[i][j];
            }
        }
    }

    return min;
}
int main ()
{
srand(time(0));
int arr[3][3];
FullMatrixWithRandomNumber(arr,3,3);
PrintMatrix(arr,3,3);
cout<<"\n\nThe maximum number is = "<<MaxNumberInMatrix(arr,3,3);
cout<<"\n\nThe minimum number is = "<<MinNumberInMatrix(arr,3,3);



    return 0;
}

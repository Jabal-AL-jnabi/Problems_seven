vector <string> Splitstring (string s1,string Delim)
{

    short pos=0;
    short count=0;
    string words;
    vector <string> VWords;
    while((pos=s1.find(Delim))!=s1.npos)
    {
        words=s1.substr(0,pos);
        VWords.push_back(words);
        s1.erase(0,pos+Delim.length());
    }

    if(s1!="")
    {
        VWords.push_back(s1);
    }
    return VWords;

}
StClient ConvertLineToRecorde (string s1)
{

    RecordeClientData.pincode=VClient[2];
    RecordeClientData.phone=VClient[3];
    RecordeClientData.AccountBalance=stod(VClient[4]);

    return RecordeClientData;

}

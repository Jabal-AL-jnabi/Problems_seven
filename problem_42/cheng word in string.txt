#include <iostream>
#include <string>
using namespace std;

void changWordInString(string &s1, string reblct,string reblaceto)
{
   short pos=s1.find(reblct);
   while(pos!=s1.npos)
   {
       s1.replace(pos,reblct.length(),reblaceto);
       pos=s1.find(reblct);
   }
    
}


int main()
{
   
  string s1="welcome to baghdad , baghdad is capital of iraq";
  string reblct="baghdad";
  string reblaceto="al mansor";
  
  changWordInString(s1,reblct,reblaceto);
  
   cout<<s1;
   
   
  
    return 0;
}
